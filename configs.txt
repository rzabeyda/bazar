# ============================================================
# /etc/systemd/system/bazar_bot.service
# ============================================================
[Unit]
Description=Egypt Bazaar Bot
After=network.target

[Service]
WorkingDirectory=/root/bazar
ExecStart=/root/bazar/venv/bin/python bot.py
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target


# ============================================================
# /etc/systemd/system/bazar_api.service
# ============================================================
[Unit]
Description=Egypt Bazaar API
After=network.target

[Service]
WorkingDirectory=/root/bazar
ExecStart=/root/bazar/venv/bin/uvicorn api:app --host 0.0.0.0 --port 8001
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target


# ============================================================
# /etc/nginx/sites-available/bazar
# ============================================================
server {
    listen 80;
    server_name egypt-bazar.com www.egypt-bazar.com;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name egypt-bazar.com www.egypt-bazar.com;

    ssl_certificate /etc/letsencrypt/live/egypt-bazar.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/egypt-bazar.com/privkey.pem;

    location / {
        proxy_pass http://127.0.0.1:8001;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        client_max_body_size 10M;
    }
}
